// ============= USER ,AUTH VIA SUPEBASE=============
export const users = pgTable(
  "users",
  {
    id: text("id")
      .primaryKey()
      .$defaultFn(() => crypto.randomUUID()),
    email: text("email").notNull().unique(),
    name: text("name"),
    avatar: text("avatar"),
role: text("role", { enum: ["USER", "ADMIN", "DEVELOP"] })
      .notNull()
      .default("USER"),
  banned: boolean("banned").default(false),
  banReason: text("ban_reason"),
  banExpires: timestamp("ban_expires"),

    createdAt: timestamp("created_at").defaultNow().notNull(),
    updatedAt: timestamp("updated_at")
      .defaultNow()
      .notNull()
      .$onUpdate(() => new Date()),
  },
);

// ============= API KEY =============
export const apiKeys = pgTable(
  "api_keys",
  {
 // Core Identity Fields
  id: text("id").primaryKey(),
  name: text("name").notNull(),

  // Key Management
  start: text("start"), // First few chars for UI display (auto-generated by Better Auth)
  prefix: text("prefix"), // Custom prefix (e.g., 'sk_live_', 'pk_test_')
  key: text("key").notNull().unique(), // Hashed API key

  // User Association
  userId: text("user_id")
    .notNull()
    .references(() => users.id, { onDelete: "cascade" }),

  // Rate Limiting (Built-in Better Auth feature)
  rateLimitEnabled: boolean("rate_limit_enabled").default(false),
  rateLimitTimeWindow: integer("rate_limit_time_window"), // Time window in ms
  rateLimitMax: integer("rate_limit_max"), // Max requests per window
  requestCount: integer("request_count").default(0), // Current request count

  // Usage Tracking & Refill System
  remaining: integer("remaining"), // Remaining requests (for quota system)
  refillAmount: integer("refill_amount"), // Amount to refill when refilling
  refillInterval: integer("refill_interval"), // Interval in ms to refill
  lastRefillAt: timestamp("last_refill_at"), // When was last refill
  lastRequest: timestamp("last_request"), // When was last API request

  // Status & Expiration
  enabled: boolean("enabled").default(true).notNull(), // Is key active/enabled
  expiresAt: timestamp("expires_at"), // Optional expiration date
  permissions: text("permissions"), // JSON string: '{record<string, string[]>}'
  
  metadata: text("metadata").$type<{
    // Key type
    scopes?: string[];

    // Security restrictions
    ipWhitelist?: string[]; // ['192.168.1.1', '10.0.0.0/8']
    allowedOrigins?: string[]; // ['https://example.com']
    allowedMethods?: Text('allowed_methods').array().default(['GET']).notNull(),

    notes?: string; // Internal notes about this key

  }>(),

  // Timestamps
  createdAt: timestamp("created_at").defaultNow().notNull(),
  updatedAt: timestamp("updated_at")
    .defaultNow()
    .notNull()
    .$onUpdate(() => new Date()),

  },
  (table) => ({
    userIdIdx: index("api_keys_user_id_idx").on(table.userId),
    enabledIdx: index("api_keys_enabled_idx").on(table.enabled),
    expiresAtIdx: index("api_keys_expires_at_idx").on(table.expiresAt),
    keyIdx: index("api_keys_key_idx").on(table.key),
  })
);

// ============= METADATA  =============
export const metadata = pgTable(
  "metadata",
  {
    id: text("id")
      .primaryKey()
      .$defaultFn(() => crypto.randomUUID()),

    // OpenSea Standard Fields
    name: text("name").notNull(),
    description: text("description"),
    symbol: text("symbol"), // Token symbol (Magic Eden/Metaplex)

    image: text("image").notNull(),
    bannerImage:text('banner_image'),
    featuredImage:text('featured_image'),

    animationUrl: text("animation_url"),
    /*
    A URL to a multi-media attachment for the item. The file extensions GLTF, GLB, WEBM, MP4, M4V, OGV, and OGG are supported, along with the audio-only extensions MP3, WAV, and OGA.

Animation_url also supports HTML pages, allowing you to build rich experiences and interactive NFTs using JavaScript canvas, WebGL, and more. Scripts and relative paths within the HTML page are now supported. However, access to browser extensions is not supported.
    */
    externalUrl: text("external_url"),
    /*
    This is the URL that will appear on OpenSea for the asset and will allow users to leave OpenSea and view the item on your site.

    */
    backgroundColor: text("background_color"),
    /*
    Background color of the item on OpenSea. Must be a six-character hexadecimal without a pre-pended #.
    */

    attributes: text("attributes").$type<{
      displayType?: "number" | "date" | "boost_number" | "boost_percentage";
      displayValue?: string;
      traitType: string;
      value: string | number;
      maxValue?: number; // For numeric traits with ceiling
    }[]>(),

    // Media info (extracted for quick filter)
    mediaType: mediaTypeEnum("media_type").default("IMAGE").notNull(),

    // IPFS Storage
    ipfsHash: text("ipfs_hash").unique(),
    ipfsUrl: text("ipfs_url"),
    isPinned: boolean("is_pinned").default(false).notNull(),
    pinnedAt: timestamp("pinned_at"),

    // Creator royalties (Magic Eden/Metaplex)
    creators: text("creators").$type<{
      address: string;
      verified: boolean;
      share: number; // Percentage share (0-100)
    }[]>(),
    sellerFeeBasisPoints: integer("seller_fee_basis_points"), // Royalty fee in basis points
    feeRecipient: text("fee_recipient"), // Royalty recipient address

    // Version control
    version: integer("version").default(1).notNull(),
    isLocked: boolean("is_locked").default(false).notNull(),

    // Timestamps
    createdAt: timestamp("created_at").defaultNow().notNull(),
    updatedAt: timestamp("updated_at")
      .defaultNow()
      .notNull()
      .$onUpdate(() => new Date()),
  },
  (table) => ({
    identifierIdx: index("metadata_identifier_idx").on(table.identifier),
    mediaTypeIdx: index("metadata_media_type_idx").on(table.mediaType),
    ipfsHashIdx: index("metadata_ipfs_hash_idx").on(table.ipfsHash),
    versionIdx: index("metadata_version_idx").on(table.version),
    isLockedIdx: index("metadata_is_locked_idx").on(table.isLocked),
    createdAtIdx: index("metadata_created_at_idx").on(table.createdAt),
  })
);

// ============= STORAGE LINKS =============
export const storageLinks = pgTable(
  "storage_links",
  {
    id: text("id")
      .primaryKey()
      .$defaultFn(() => crypto.randomUUID()),

    // Link to media or metadata
    mediaId: text("media_id").references(() => media.id, { onDelete: "cascade" }),
    metadataId: text("metadata_id").references(() => metadata.id, { onDelete: "cascade" }),

    // Storage Provider Info
    provider: storageProviderEnum("provider").notNull(),
    status: storageStatusEnum("status").default("ACTIVE").notNull(),

    // Storage URLs
    url: text("url").notNull(), // Primary URL for this storage
    publicUrl: text("public_url"), // Public facing URL (if different)
    cdnUrl: text("cdn_url"), // CDN URL (if applicable)

    // IPFS/Pinata specific
    ipfsHash: text("ipfs_hash"), // CID for IPFS
    ipfsGateway: text("ipfs_gateway"), // e.g., 'https://gateway.pinata.cloud/ipfs/'
    isPinned: boolean("is_pinned").default(false),
    pinnedAt: timestamp("pinned_at"),

    // Supabase specific
    bucketName: text("bucket_name"), // Supabase bucket name
    objectPath: text("object_path"), // Path within bucket

    // Provider metadata
    providerMetadata: text("provider_metadata").$type<{
      // Supabase
      bucketId?: string;
      publicBucket?: boolean;
      
      // Pinata
      pinataId?: string;
      pinataName?: string;
      
      // ImageKit
      imageKitId?: string;
      fileId?: string;
      
      // Generic
      etag?: string;
      lastModified?: string;
      contentType?: string;
      size?: number;
      
      // Additional provider-specific fields
      [key: string]: any;
    }>(),

    uploadedAt: timestamp("uploaded_at").defaultNow().notNull(),

    // Verification & Health
    isVerified: boolean("is_verified").default(false).notNull(), // Has URL been verified accessible?
    lastVerifiedAt: timestamp("last_verified_at"),
    failureCount: integer("failure_count").default(0).notNull(),
    lastFailureAt: timestamp("last_failure_at"),
    lastFailureReason: text("last_failure_reason"),

    // Soft delete
    deletedAt: timestamp("deleted_at"),

    createdAt: timestamp("created_at").defaultNow().notNull(),
    updatedAt: timestamp("updated_at")
      .defaultNow()
      .notNull()
      .$onUpdate(() => new Date()),
  },
  (table) => ({
    mediaIdIdx: index("storage_links_media_id_idx").on(table.mediaId),
    metadataIdIdx: index("storage_links_metadata_id_idx").on(table.metadataId),
    providerIdx: index("storage_links_provider_idx").on(table.provider),
    statusIdx: index("storage_links_status_idx").on(table.status),
    ipfsHashIdx: index("storage_links_ipfs_hash_idx").on(table.ipfsHash),
  })
);

export const auditLogs = pgTable(
  "audit_logs",
  {
    id: uuid("id").primaryKey().defaultRandom(),

    // Can be from session or API key
    userId: text("user_id").references(() => users.id, { onDelete: "set null" }),
    apiKeyId: text("api_key_id").references(() => apiKeys.id, {
      onDelete: "set null",
    }),

    method: varchar("method", { length: 10 }).notNull(), // GET, POST, PUT, DELETE
    path: varchar("path", { length: 500 }).notNull(),
    action: varchar("action", { length: 100 }).notNull(),

    ipAddress: varchar("ip_address", { length: 45 }),
    userAgent: varchar("user_agent", { length: 500 }),

    resourceType: varchar("resource_type", { length: 50 }),
    resourceId: uuid("resource_id"),

    statusCode: integer("status_code").notNull(),
    duration: integer("duration"), // milliseconds

    metadata: jsonb("metadata").$type<{
      error?: string;
      requestBody?: Record<string, unknown>;
      responseSize?: number;
    }>(),

    createdAt: timestamp("created_at").defaultNow().notNull(),
  },
  (table) => ({
    userIdIdx: index("audit_logs_user_id_idx").on(table.userId),
    apiKeyIdIdx: index("audit_logs_api_key_id_idx").on(table.apiKeyId),
    createdAtIdx: index("audit_logs_created_at_idx").on(table.createdAt),
    actionIdx: index("audit_logs_action_idx").on(table.action),
    resourceTypeIdx: index("audit_logs_resource_type_idx").on(table.resourceType),
  })
);

// ============= RELATIONS =============
export const usersRelations = relations(users, ({ many }) => ({
  apiKeys: many(apiKeys),
  auditLogs: many(auditLogs),
}));

export const apiKeysRelations = relations(apiKeys, ({ one, many }) => ({
  user: one(users, {
    fields: [apiKeys.userId],
    references: [users.id],
  }),
  auditLogs: many(auditLogs),
}));

export const metadataRelations = relations(metadata, ({ many }) => ({
  storageLinks: many(storageLinks),
}));

export const mediaRelations = relations(media, ({ many }) => ({
  storageLinks: many(storageLinks),
}));

export const storageLinksRelations = relations(storageLinks, ({ one }) => ({
  media: one(media, {
    fields: [storageLinks.mediaId],
    references: [media.id],
  }),
  metadata: one(metadata, {
    fields: [storageLinks.metadataId],
    references: [metadata.id],
  }),
}));

export const auditLogsRelations = relations(auditLogs, ({ one }) => ({
  user: one(users, {
    fields: [auditLogs.userId],
    references: [users.id],
  }),
  apiKey: one(apiKeys, {
    fields: [auditLogs.apiKeyId],
    references: [apiKeys.id],
  }),
}));
